\newglossaryentry{Naive Bayes}
{
    name=Naive Bayes,
    description={Naive Bayes is a probabilistic classifier based on Bayes' theorem. It assumes that features are conditionally independent given the class label, hence the term "naive." It calculates the probability of an instance belonging to each class and assigns it to the class with the highest probability. Naive Bayes is computationally efficient and works well with large feature spaces. It is commonly used in text classification and spam filtering.}
}

\newglossaryentry{J48}
{
name=J48,
description={J48, also known as C4.5, is a decision tree classifier developed by Ross Quinlan. It constructs a tree-based model by recursively splitting the data based on the most informative features. The splits are determined using entropy or information gain measures. J48 handles both categorical and numerical data, and it can handle missing values by employing various techniques. Decision trees like J48 provide interpretable models, and they are widely used in data mining and knowledge discovery tasks.}
}

\newglossaryentry{k-NN}
{
name=k-Nearest Neighbors,
description={k-Nearest Neighbors is a non-parametric instance-based classification algorithm. It assigns a class label to an instance based on the majority vote of its k nearest neighbors in the feature space. The "k" represents the number of neighbors to consider. The distance metric (e.g., Euclidean, Manhattan) is used to determine the nearest neighbors. k-NN is simple to understand and implement, and it can be effective for certain types of data. It is used in various domains such as pattern recognition, recommendation systems, and anomaly detection.}
}

\newglossaryentry{SVM}
{
name=Support Vector Machines,
description={Support Vector Machines is a powerful and versatile supervised learning algorithm that performs both classification and regression tasks. It finds an optimal hyperplane in a high-dimensional feature space that maximally separates the data points of different classes. SVM can handle linearly separable as well as non-linearly separable data by using kernel functions (e.g., linear, polynomial, radial basis function). SVM is known for its ability to handle complex datasets and generalization to unseen data. It is widely used in image classification, text categorization, and bioinformatics.}
}
